Experiment 1: Central Tendency and Dispersion

Aim: To write a Python program to compute Mean, Median, Mode, Variance and Standard Deviation.

Program Using Statistics Module

import statistics

# Input data
data = list(map(float, input("Enter the data values separated by spaces: ").split()))

# Mean
mean = statistics.mean(data)

# Median
median = statistics.median(data)

# Mode
try:
    mode = statistics.mode(data)
except statistics.StatisticsError:
    mode = "No unique mode"

# Variance (Sample Variance)
variance = statistics.variance(data)

# Standard Deviation (Sample Standard Deviation)
std_deviation = statistics.stdev(data)

# Display Results
print("\n----- Statistical Measures -----")
print("Mean               :", mean)
print("Median             :", median)
print("Mode               :", mode)
print("Variance           :", variance)
print("Standard Deviation :", std_deviation)

Sample Input

Enter the data values separated by spaces: 10 20 30 40 50

Sample Output

Manual Implementation Without Statistics Module

import math
from collections import Counter

# Input data
data = list(map(float, input("Enter the data values separated by spaces: ").split()))
n = len(data)

# Mean
mean = sum(data) / n

# Median
sorted_data = sorted(data)
if n % 2 == 0:
    median = (sorted_data[n//2 - 1] + sorted_data[n//2]) / 2
else:
    median = sorted_data[n//2]

# Mode
count = Counter(data)
max_freq = max(count.values())
modes = [key for key, value in count.items() if value == max_freq]

if max_freq == 1:
    mode = "No Mode"
elif len(modes) == 1:
    mode = modes[0]
else:
    mode = modes

# Sample Variance
variance = sum((x - mean) ** 2 for x in data) / (n - 1)

# Sample Standard Deviation
std_deviation = math.sqrt(variance)

# Output
print("\nMean               :", mean)
print("Median             :", median)
print("Mode               :", mode)
print("Variance           :", variance)
print("Standard Deviation :", std_deviation)

Important Functions

MeasurePurpose
MeanFinds the average value of the data.
MedianFinds the middle value after sorting the data.
ModeFinds the most frequently occurring value.
VarianceMeasures how spread out the values are from the mean.
Standard DeviationMeasures the average spread of data around the mean.

Experiment 2: Study of Basic Python Libraries

Aim: To study the basic Python libraries Statistics, Math, NumPy and SciPy.

Python Program

import statistics
import math
import numpy as np
from scipy import stats

# Sample Data
data = [10, 20, 30, 40, 50]

print("Original Data:", data)

# ---------------- Statistics Library ----------------
print("\n--- Statistics Library ---")
print("Mean               :", statistics.mean(data))
print("Median             :", statistics.median(data))
print("Mode               :", statistics.mode(data))
print("Variance           :", statistics.variance(data))
print("Standard Deviation :", statistics.stdev(data))

# ---------------- Math Library ----------------
print("\n--- Math Library ---")
print("Square Root of 25  :", math.sqrt(25))
print("Factorial of 5     :", math.factorial(5))
print("Power (2^5)        :", math.pow(2, 5))
print("Value of Pi        :", math.pi)
print("Value of e         :", math.e)
print("Log(10)            :", math.log(10))
print("Sin(30°)           :", math.sin(math.radians(30)))

# ---------------- NumPy Library ----------------
print("\n--- NumPy Library ---")
arr = np.array(data)

print("Array              :", arr)
print("Mean               :", np.mean(arr))
print("Median             :", np.median(arr))
print("Sum                :", np.sum(arr))
print("Maximum            :", np.max(arr))
print("Minimum            :", np.min(arr))
print("Standard Deviation :", np.std(arr))
print("Square of Array    :", np.square(arr))

# ---------------- SciPy Library ----------------
print("\n--- SciPy Library ---")
print("Mean               :", stats.tmean(arr))
print("Mode               :", stats.mode(arr, keepdims=True))
print("Skewness           :", stats.skew(arr))
print("Kurtosis           :", stats.kurtosis(arr))
print("Z-Score            :", stats.zscore(arr))

Experiment 3: Pandas and Matplotlib for ML

Aim: To study Python libraries for Machine Learning applications such as Pandas and Matplotlib.

Python Program

import pandas as pd
import matplotlib.pyplot as plt

# Create a DataFrame
data = {
    "Student": ["A", "B", "C", "D", "E"],
    "Marks": [85, 90, 78, 92, 88],
    "Age": [20, 21, 20, 22, 21]
}

df = pd.DataFrame(data)

# ---------------- Pandas Operations ----------------
print("Original DataFrame")
print(df)

print("\nFirst 3 Rows")
print(df.head(3))

print("\nStatistical Summary")
print(df.describe())

print("\nAverage Marks:", df["Marks"].mean())
print("Maximum Marks:", df["Marks"].max())
print("Minimum Marks:", df["Marks"].min())

# ---------------- Matplotlib Visualization ----------------

# Bar Chart
plt.figure(figsize=(6,4))
plt.bar(df["Student"], df["Marks"])
plt.title("Student Marks")
plt.xlabel("Students")
plt.ylabel("Marks")
plt.grid(True)
plt.show()

# Line Chart
plt.figure(figsize=(6,4))
plt.plot(df["Student"], df["Marks"], marker='o')
plt.title("Student Marks Trend")
plt.xlabel("Students")
plt.ylabel("Marks")
plt.grid(True)
plt.show()

# Pie Chart
plt.figure(figsize=(6,6))
plt.pie(df["Marks"], labels=df["Student"], autopct="%1.1f%%")
plt.title("Marks Distribution")
plt.show()
